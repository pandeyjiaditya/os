//process 2
#include <iostream>
#include <string>
#include <fstream>
#include <sstream>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <errno.h>

using namespace std;

int main()
{
    string fifo1 = "fifo1";
    string fifo2 = "fifo2";

    // Create FIFOs if they do not exist
    if (mkfifo(fifo1.c_str(), 0666) == -1 && errno != EEXIST)
    {
        perror("mkfifo fifo1");
        return 1;
    }

    if (mkfifo(fifo2.c_str(), 0666) == -1 && errno != EEXIST)
    {
        perror("mkfifo fifo2");
        return 1;
    }

    cout << "Process 2 Started" << endl;

    // Open first FIFO for reading
    int fd1 = open(fifo1.c_str(), O_RDONLY);

    if (fd1 == -1)
    {
        perror("open fifo1");
        return 1;
    }

    string data;
    char buffer[1024];
    int bytes;

    // Read sentences from Process 1
    while ((bytes = read(fd1, buffer, sizeof(buffer))) > 0)
    {
        data.append(buffer, bytes);
    }

    close(fd1);

    // Count characters
    int characters = data.size();

    // Count words
    int words = 0;
    string word;
    stringstream ss(data);

    while (ss >> word)
    {
        words++;
    }

    // Count lines
    int lines = 0;
    string line;
    stringstream ss2(data);

    while (getline(ss2, line))
    {
        lines++;
    }

    // Write results into text file
    ofstream file("output.txt");

    file << "----- Sentence Analysis -----" << endl;
    file << "Number of Characters: " << characters << endl;
    file << "Number of Words: " << words << endl;
    file << "Number of Lines: " << lines << endl;

    file.close();

    cout << "Analysis completed and saved in output.txt" << endl;

    // Read contents of output file
    ifstream inputFile("output.txt");

    string result(
        (istreambuf_iterator<char>(inputFile)),
        istreambuf_iterator<char>()
    );

    inputFile.close();

    // Open second FIFO for writing
    int fd2 = open(fifo2.c_str(), O_WRONLY);

    if (fd2 == -1)
    {
        perror("open fifo2");
        return 1;
    }

    // Send file contents to Process 1
    write(fd2, result.c_str(), result.size());

    close(fd2);

    cout << "Result sent to Process 1." << endl;

    return 0;
}
