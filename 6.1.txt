//process 1
#include <iostream>
#include <string>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <errno.h>

using namespace std;

int main()
{
    string fifo1 = "fifo1";
    string fifo2 = "fifo2";

    // Create two FIFOs
    if (mkfifo(fifo1.c_str(), 0666) == -1 && errno != EEXIST)
    {
        perror("mkfifo fifo1");
        return 1;
    }

    if (mkfifo(fifo2.c_str(), 0666) == -1 && errno != EEXIST)
    {
        perror("mkfifo fifo2");
        return 1;
    }

    cout << "Process 1 Started" << endl;

    // Open first FIFO for writing
    int fd1 = open(fifo1.c_str(), O_WRONLY);

    if (fd1 == -1)
    {
        perror("open fifo1");
        return 1;
    }

    cout << "Enter sentences (type END to finish):" << endl;

    string sentence;

    while (true)
    {
        getline(cin, sentence);

        if (sentence == "END")
            break;

        sentence += "\n";

        write(fd1, sentence.c_str(), sentence.size());
    }

    // Close first FIFO
    close(fd1);

    cout << "\nWaiting for Process 2 response..." << endl;

    // Open second FIFO for reading
    int fd2 = open(fifo2.c_str(), O_RDONLY);

    if (fd2 == -1)
    {
        perror("open fifo2");
        return 1;
    }

    cout << "\n----- Result Received -----\n";

    char buffer[1024];
    int bytes;

    while ((bytes = read(fd2, buffer, sizeof(buffer))) > 0)
    {
        cout.write(buffer, bytes);
    }

    cout << endl;

    close(fd2);

    cout << "Process 1 Completed." << endl;

    return 0;
}
